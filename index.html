// Handle Approve / Reject Action on leave_requests table and send notification email to staff async function updateLeaveRequestStatus(requestId, newStatus) { try { // Find the leave request record to get staff details (sdms_id, dates, reason) const leaveReq = cachedLeaves.find(r => r.id === requestId); const staffObj = leaveReq ? cachedStaff.find(s => s.sdms_id === leaveReq.sdms_id) : null; const { error } = await supabaseClient .from('leave_requests') .update({ status: newStatus }) .eq('id', requestId); if (error) throw error; // Send notification email to the staff member if email exists if (staffObj && staffObj.email) { const staffName = `${staffObj.first_name || ''} ${staffObj.last_name || ''}`.trim(); const subject = `Leave Request Status Update: ${newStatus.toUpperCase()}`; const body = `Dear ${staffName},\n\n` + `Your leave request for the date: ${leaveReq.leave_date || leaveReq.start_date || 'N/A'}\n` + `Reason: ${leaveReq.reason || 'N/A'}\n\n` + `Has been officially reviewed and its status has been updated to: ${newStatus.toUpperCase()}.\n\n` + `Best regards,\n` + `GS Gikagati Administration\n` + `Head Teacher: Ladislas`; const gmailUrl = `https://mail.google.com/mail/?view=cm&fs=1&to=${encodeURIComponent(staffObj.email)}&su=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`; window.open(gmailUrl, '_blank'); } else if (staffObj && !staffObj.email) { console.warn("Staff member found, but no email address is registered in cachedStaff."); } alert(`Leave request successfully ${newStatus}!`); loadDashboardData(); } catch (err) { alert(`Error updating leave request status: ${err.message}`); console.error(err); } }
